function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  // --- ฟังก์ชันเสริม: ตรวจสอบสร้างชีต และสร้างหัวคอลัมน์อัตโนมัติ ---
  function getOrCreateSheet(sheetName) {
    var sheet = ss.getSheetByName(sheetName);
    if (sheet == null) {
      // ถ้าไม่มีชีตชื่อนี้ ให้สร้างขึ้นมาใหม่
      sheet = ss.insertSheet(sheetName);
      
      // เมื่อสร้างชีตใหม่ ให้ใส่หัวคอลัมน์ 5 คอลัมน์ (เพิ่ม Scale Name)
      sheet.appendRow(["Timestamp", "Target weight", "Resulting weight", "Actual weight", "Scale Name"]);
      
      // ทำให้หัวคอลัมน์ A1-E1 เป็นตัวหนา และแช่แข็งแถวแรกไว้
      sheet.getRange("A1:E1").setFontWeight("bold");
      sheet.setFrozenRows(1);
    }
    return sheet;
  }
  // ------------------------------------------

  // เรียกใช้ฟังก์ชันสร้างชีตหลักที่จะรวมข้อมูลทั้งหมด
  var mainSheet = getOrCreateSheet("รวมข้อมูลตาชั่ง");
  
  var data = JSON.parse(e.postData.contents);
  var timestamp = new Date();
  
  // 1. ดึงข้อมูล Mode Timer  LS9
  var d5032 = data.D5032;
  var d102 = (data.D102 !== undefined) ? "'" + data.D102 : undefined; 
  var d5040 = data.D5040;
  
  // 2. ดึงข้อมูล Mode Timer  UD1
  var d5048 = data.D5048;
  var d118 = (data.D118 !== undefined) ? "'" + data.D118 : undefined; 
  var d5052 = data.D5052;

  // 3. ดึงข้อมูล Mode Timer  LAS
  var d5060 = data.D5060;
  var d134 = (data.D134 !== undefined) ? "'" + data.D134 : undefined; 
  var d5064 = data.D5064;

  // 4. ดึงข้อมูล Mode Timer  CAUSTIC SODA 50%
  var d5072 = data.D5072;
  var d150 = (data.D150 !== undefined) ? "'" + data.D150 : undefined; 
  var d5076 = data.D5076;

  // 5. ดึงข้อมูล "Mode Weight  CAUSTIC SODA 50%" (แก้ไขชื่อตัวแปรให้ถูกต้องแล้ว)
  var d5108 = data.D5108;
  var d178 = (data.D178 !== undefined) ? "'" + data.D178 : undefined; 
  var d5076_weight = data.D5076_weight; 

  // --- บันทึกข้อมูลลงชีตเดียว และเพิ่มชื่อตาชั่งต่อท้าย ---

  // รายการบันทึก Line Protocol ส่งไปยัง InfluxDB ควบคู่กัน (Dual-Write)
  var influxLines = [];

  // 1. ข้อมูล "Mode Timer  LS9"
  if (d5032 !== undefined || data.D102 !== undefined || d5040 !== undefined) {
    mainSheet.appendRow([timestamp, d5032, d102, d5040, "Mode Timer  LS9"]);
    influxLines.push(formatLineProtocol("Mode Timer  LS9", d5032, data.D102, d5040, timestamp));
  }

  // 2. ข้อมูล "Mode Timer  UD1"
  if (d5048 !== undefined || data.D118 !== undefined || d5052 !== undefined) {
    mainSheet.appendRow([timestamp, d5048, d118, d5052, "Mode Timer  UD1"]); 
    influxLines.push(formatLineProtocol("Mode Timer  UD1", d5048, data.D118, d5052, timestamp));
  }

  // 3. ข้อมูล "Mode Timer  LAS"
  if (d5060 !== undefined || data.D134 !== undefined || d5064 !== undefined) {
    mainSheet.appendRow([timestamp, d5060, d134, d5064, "Mode Timer  LAS"]); 
    influxLines.push(formatLineProtocol("Mode Timer  LAS", d5060, data.D134, d5064, timestamp));
  }

  // 4. ข้อมูล "Mode Timer  CAUSTIC SODA 50%" 
  if (d5072 !== undefined || data.D150 !== undefined || d5076 !== undefined) {
    mainSheet.appendRow([timestamp, d5072, d150, d5076, "Mode Timer  CAUSTIC SODA 50%"]); 
    influxLines.push(formatLineProtocol("Mode Timer  CAUSTIC SODA 50%", d5072, data.D150, d5076, timestamp));
  }

  // 5. ข้อมูล "Mode Weight  CAUSTIC SODA 50%"
  if (d5108 !== undefined || data.D178 !== undefined || d5076_weight !== undefined) {
    mainSheet.appendRow([timestamp, d5108, d178, d5076_weight, "Mode Weight  CAUSTIC SODA 50%"]); 
    influxLines.push(formatLineProtocol("Mode Weight  CAUSTIC SODA 50%", d5108, data.D178, d5076_weight, timestamp));
  }

  // ส่งบันทึกเข้า InfluxDB Cloud ทันที (Dual-Write)
  if (influxLines.length > 0) {
    sendBatchToInflux(influxLines);
  }

  return ContentService.createTextOutput(JSON.stringify({
    "result": "success", 
    "timestamp": timestamp
  })).setMimeType(ContentService.MimeType.JSON);
}

// --- ฟังก์ชันจัดการเกณฑ์ % ผลต่าง (Tolerance %) ในชีต Config และ ScriptProperties ---
function getStoredTolerance() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var configSheet = ss.getSheetByName("Config");
  if (!configSheet) {
    configSheet = ss.insertSheet("Config");
    configSheet.appendRow(["Setting", "Value", "Description"]);
    configSheet.appendRow(["TolerancePercent", 1.0, "เกณฑ์ % ผลต่างที่ยอมรับได้ (เช่น 1.0 คือ +-1.0%)"]);
    configSheet.getRange("A1:C1").setFontWeight("bold");
    configSheet.setFrozenRows(1);
    return 1.0;
  }
  var val = parseFloat(configSheet.getRange("B2").getValue());
  return (!isNaN(val) && val > 0) ? val : 1.0;
}

function setStoredTolerance(val) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var configSheet = ss.getSheetByName("Config");
  if (!configSheet) {
    configSheet = ss.insertSheet("Config");
    configSheet.appendRow(["Setting", "Value", "Description"]);
    configSheet.appendRow(["TolerancePercent", val, "เกณฑ์ % ผลต่างที่ยอมรับได้ (เช่น 1.0 คือ +-1.0%)"]);
    configSheet.getRange("A1:C1").setFontWeight("bold");
    configSheet.setFrozenRows(1);
  } else {
    configSheet.getRange("B2").setValue(val);
  }
  PropertiesService.getScriptProperties().setProperty("tolerancePercent", String(val));
}

// --- ฟังก์ชันสำหรับส่งข้อมูล JSON ให้กับหน้าเว็บภายนอก (GitHub Pages) ---
function doGet(e) {
  var action = (e && e.parameter) ? e.parameter.action : null;
  var callback = (e && e.parameter) ? e.parameter.callback : null;

  // 1. ตรวจสอบการตั้งค่าเกณฑ์ Tolerance % (เมื่อมีคนเปลี่ยนจากหน้าเว็บ)
  if (action === "setTolerance") {
    var val = parseFloat(e.parameter.val);
    if (!isNaN(val) && val > 0) {
      setStoredTolerance(val);
      var resObj = { result: "success", tolerancePercent: val };
      var resJson = JSON.stringify(resObj);
      if (callback) {
        return ContentService.createTextOutput(callback + '(' + resJson + ')')
            .setMimeType(ContentService.MimeType.JAVASCRIPT);
      }
      return ContentService.createTextOutput(resJson)
          .setMimeType(ContentService.MimeType.JSON);
    }
  }

  // 2. ส่งข้อมูลแดชบอร์ดตามปกติ (รวม rows และเกณฑ์ tolerancePercent ล่าสุด)
  var data = getDashboardData();
  var output = JSON.stringify(data);
  
  // รองรับ JSONP เผื่อกรณีเรียกข้ามโดเมนแบบพิเศษ
  if (callback) {
    return ContentService.createTextOutput(callback + '(' + output + ')')
        .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }
  
  // ส่งออกข้อมูลในรูปแบบ JSON
  return ContentService.createTextOutput(output)
      .setMimeType(ContentService.MimeType.JSON);
}

// --- ฟังก์ชันสำหรับดึงข้อมูลจากชีตมาส่งให้หน้า Dashboard ---
function getDashboardData() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("รวมข้อมูลตาชั่ง");
  var tolerance = getStoredTolerance();
  
  if (!sheet) {
    return { rows: [], tolerancePercent: tolerance };
  }
  
  var lastRow = sheet.getLastRow();
  if (lastRow <= 1) {
    return { rows: [], tolerancePercent: tolerance };
  }
  
  // ดึงข้อมูลทั้งหมดที่มีในชีต
  var data = sheet.getDataRange().getValues();
  
  // แปลงวันที่ให้อยู่ในรูป ISO String หรือข้อความ เพื่อส่งผ่านไปยังหน้าเว็บได้อย่างเสถียร
  var formattedData = data.map(function(row, idx) {
    if (idx === 0) return row; // แถวหัวตารางคงเดิม
    var dateVal = row[0];
    var dateStr = (dateVal instanceof Date) ? dateVal.toISOString() : (dateVal ? String(dateVal) : "");
    return [
      dateStr,
      row[1],
      row[2],
      row[3],
      row[4] ? String(row[4]) : ""
    ];
  });
  
  return {
    rows: formattedData,
    tolerancePercent: tolerance
  };
}

function formatLineProtocol(scaleName, targetVal, resultingVal, actualVal, dateObj) {
  function cleanNum(val) {
    if (val === undefined || val === null || val === '') return NaN;
    var s = String(val).replace(/^['"\s]+/, '').replace(/,/g, '').trim();
    var n = parseFloat(s);
    return isNaN(n) ? NaN : n;
  }

  var chem = "Other";
  if (scaleName.indexOf("CAUSTIC") > -1) chem = "CAUSTIC SODA 50%";
  else if (scaleName.indexOf("LS9") > -1) chem = "LS9";
  else if (scaleName.indexOf("UD1") > -1) chem = "UD1";
  else if (scaleName.indexOf("LAS") > -1) chem = "LAS";

  var mode = "Unknown";
  if (scaleName.indexOf("Mode Weight") > -1) mode = "Mode Weight";
  else if (scaleName.indexOf("Mode Timer") > -1) mode = "Mode Timer";

  var t = cleanNum(targetVal);
  var r = cleanNum(resultingVal);
  var a = cleanNum(actualVal);

  var effA = (!isNaN(a) && a !== 0) ? a : ((!isNaN(r) && r !== 0) ? r : 0);
  var effT = !isNaN(t) ? t : 0;
  var effR = !isNaN(r) ? r : 0;

  function escapeTag(str) {
    return String(str || '').replace(/,/g, '\\,').replace(/ /g, '\\ ').replace(/=/g, '\\=');
  }

  var tags = "chemical=" + escapeTag(chem) + ",mode=" + escapeTag(mode) + ",scale_name=" + escapeTag(scaleName);
  var fields = "target=" + effT + ",resulting=" + effR + ",actual=" + effA;
  var epochSec = Math.floor(dateObj.getTime() / 1000);

  return "scale_dosing," + tags + " " + fields + " " + epochSec;
}

function sendBatchToInflux(lines) {
  if (!lines || lines.length === 0) return;
  try {
    var writeUrl = INFLUX_CONFIG.url + "/api/v2/write?org=" + encodeURIComponent(INFLUX_CONFIG.org) + 
                   "&bucket=" + encodeURIComponent(INFLUX_CONFIG.bucket) + "&precision=s";
    var options = {
      method: "post",
      contentType: "text/plain; charset=utf-8",
      headers: {
        "Authorization": "Token " + INFLUX_CONFIG.token
      },
      payload: lines.join("\n"),
      muteHttpExceptions: true
    };
    var response = UrlFetchApp.fetch(writeUrl, options);
    var code = response.getResponseCode();
    if (code !== 204) {
      Logger.log("InfluxDB write error HTTP " + code + ": " + response.getContentText());
    }
  } catch (err) {
    Logger.log("InfluxDB write exception: " + err.message);
  }
}

// --- ฟังก์ชันย้าย/ซิงค์ข้อมูลประวัติเก่าทั้งหมดจาก Google Sheet เข้า InfluxDB ---
function migrateSheetToInfluxDB() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName("รวมข้อมูลตาชั่ง");
  if (!sheet) {
    Logger.log("ไม่พบชีต 'รวมข้อมูลตาชั่ง'");
    return;
  }
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) {
    Logger.log("ไม่มีข้อมูลในชีต");
    return;
  }

  var lines = [];
  var totalMigrated = 0;

  for (var i = 1; i < data.length; i++) {
    var row = data[i];
    var dateVal = row[0];
    var dateObj = (dateVal instanceof Date) ? dateVal : new Date(dateVal);
    if (isNaN(dateObj.getTime())) continue;

    var target = row[1];
    var resulting = row[2];
    var actual = row[3];
    var scaleName = String(row[4] || '').trim();
    if (!scaleName) continue;

    var line = formatLineProtocol(scaleName, target, resulting, actual, dateObj);
    lines.push(line);

    if (lines.length >= 200) {
      sendBatchToInflux(lines);
      totalMigrated += lines.length;
      lines = [];
      Utilities.sleep(100);
    }
  }

  if (lines.length > 0) {
    sendBatchToInflux(lines);
    totalMigrated += lines.length;
  }

  Logger.log("ย้ายข้อมูลเข้า InfluxDB สำเร็จทั้งหมด: " + totalMigrated + " รายการ");
}
